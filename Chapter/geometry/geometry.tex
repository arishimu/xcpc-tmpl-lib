\section{计算几何}

\begin{enumerate}[noitemsep]
    \item {\bf 向量}：
    \begin{itemize}[noitemsep]
        \item 向量加减、数乘；
        \item 向量的内积、外积；
        \item 向量的模长、单位向量；
        \item 向量的 $\pm \pi / 2$ 旋转、$\theta$ 旋转；
        \item 向量的夹角、$[-\pi, \pi]$ 有向角、$[0, 2\pi]$ 有向角；
    \end{itemize}
    \item {\bf 点}：
    \begin{itemize}[noitemsep]
        \item 点所处半平面的符号；
        \item 两点之间的重合判定；
        \item 两点之间的 Eculid 距离；
        \item 三点之间的逆时针/顺时针关系判定；
        \item 三点之间的共线判定；
        \item 点是否在角内的判定；
        \item 点的字典序、方向序、极角序；
        \item 点绕中心的 $\pm \pi / 2$ 旋转、$\theta$ 旋转；
    \end{itemize}
    \item {\bf 直线}：
    \begin{itemize}[noitemsep]
        \item 直线的定义 $v \times P = c$；
        \item 直线的两点式、点向式、点斜式、一般式；
        \item 直线的平移构造、平行构造、垂直构造；
    \end{itemize}
    \item {\bf 点与直线}：
    \begin{itemize}[noitemsep]
        \item 点是否在直线上的判定；
        \item 点到直线的带符号半平面距离、距离、带符号距离；
        \item 点到直线的垂足、对称点；
    \end{itemize}
    \item {\bf 直线与直线}：
    \begin{itemize}[noitemsep]
        \item 两条直线的重合、平行、垂直、相交判定；
        \item 两条直线的交点；
    \end{itemize}
    \item {\bf 点与线段}：
    \begin{itemize}[noitemsep]
        \item 点是否在直径确定的圆内的判定；
        \item 点是否在线段上的判定；
        \item 点到线段的距离；
    \end{itemize}
    \item {\bf 线段与线段}：
    \begin{itemize}[noitemsep]
        \item 两条线段完美相交的判定；
        \item 两条线段的交点、交线段；
        \item 两条线段之间的距离；
    \end{itemize}
    \item {\bf 圆}：
    \begin{itemize}[noitemsep]
        \item 圆的三点构造；
        \item 圆上取点；
        \item 过圆外一点作圆的切线；
        \item 直线与圆的交点；
        \item 圆与圆的交点；
    \end{itemize}
    \item {\bf 多边形}：
    \begin{itemize}[noitemsep]
        \item 多边形的面积；
        \item 点是否在多边形上/内的判定；
        \item 点是否在凸多边形内的判定；
        \item 点集的凸包；
    \end{itemize}
    \item {\bf 其他}：
    \begin{itemize}[noitemsep]
        \item 点集内的最近点对；
    \end{itemize}
\end{enumerate}
